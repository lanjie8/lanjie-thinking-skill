/**
 * code组件使用的回调函数
 * @param datas 传递回来的数据，json格式
 */
function codeCallBack(datas) {
    if (typeof datas === 'string') {
        try {
            datas = JSON.parse(datas);
        } catch (e) {
            return;
        }
    }

    if (datas.method == "sendCodeValue") {
        var codeid = datas.params.codeid;
        var codeName = datas.params.codeName;
        var formx = $("#isFmVisibleForm").parents("form");
        $(formx).find("input[codeid='"+codeid+"']").each(function(){
            $(this).remove();
        });
        var dmmcView = "";

        dmmcView = dmmcView + '<li class="search-choice" data-choiceid="'+codeid+'" data-choicename="'+codeName+'">';
        dmmcView = dmmcView + '<span id="code_'+codeid+'" name="code_'+codeid+'">'+datas.params.dmmc+'</span>';
        dmmcView = dmmcView + '<a class="search-choice-close" onClick="clearInputCodes(this)"></a></li>';
        $("#"+escapeJquery(codeid)+"_chosen").find("ul").html(dmmcView);
        $("#" + escapeJquery(codeid)).val(datas.params.dmbs);
        var input = $("<input type='hidden'>");
        $(input).attr("name", codeName);
        $(input).attr("id", codeid);
        $(input).attr("codeid", codeid);
        $(input).val(datas.params.dmbs);
        $("#"+escapeJquery(codeid)+"_chosen").after($(input));
        var callBackName = datas.params.callBack;
        if(callBackName && callBackName!==""){
            var func = callBackName+'("'+codeid+'","'+datas.params.dmbs+'")'
            eval(func)
        }
    }
}

function openCodeWindow(obj,labelx,framepath,name,useCommon,isTable,codeid){
    var value = new Array();
    var para = $(obj).attr("parameters");
    var paraEval = eval("(" + para + ")");
    $("input[codeid='"+codeid+"']").each(function(){
        value.push($(this).val());
    });
    //动态行里trt处理
    paraEval.codeid=codeid;
    paraEval.codeName=name;
    paraEval.framepath=framepath;
    if(value){
        paraEval.selectedVal=JSON.stringify(value);
    }
    para = JSON.stringify(paraEval);
    $(obj).attr("parameters",para);
    if(useCommon === "true" || isTable==="true"){
        openWindow(obj,{ size:'large',title:labelx,framepath:framepath});
    }else {
        openWindow(obj,{size:"small",title:labelx,framepath:framepath});
    }
}

function selectLine(id,name,obj,type,showtitle){
    var xzz=[];
    var namexzz=[];
    if($(obj).parent().attr("class").indexOf("select")>-1) {
        $(obj).parent().attr("class", "item");
    }else{
        $(obj).parent().attr("class", "item select");
    }
    if(type=="all"){
        $("#"+escapeJquery(id)).val("");
    }else if(type=="single"){
        if($(obj).parent().attr("class").indexOf("select")>-1&&$(obj).parent().attr("value")!="") {
            xzz.push($(obj).parent().attr("value"));
            namexzz.push($(obj).parent().attr("name"));

    }else{
        $("#"+escapeJquery(id)+"_dd div").each(function(){
            if($(this).attr("class").indexOf("select")>-1&&$(this).attr("value")!=""){
                xzz.push($(this).attr("value"));
                namexzz.push($(this).attr("name"));
            }
            $("#"+escapeJquery(id)).val(xzz);
        });

            showText = titleContent + "(" + namexzz.join(",").replaceAll(";","") + ")";
        }
        document.getElementById(titleid).innerText = showText;
    }

//表格带有合并行的间隔变色处理
function rowspanStriped(datatable){
    var stable = $(datatable);
    var rowspan = 0;
    var iscolor = false;
    stable.find("tbody").find("tr").each(function () {
        var attr = $(this).find('td').first().attr("rowspan");
        if (typeof attr !== typeof undefined && attr !== false) {
            rowspan = attr;
        }
        if (rowspan > 0) {
            if (iscolor == true) {
            $(this).addClass("tdbg");
            }
            rowspan--;
            if (rowspan == 0) {
                iscolor = (iscolor == true) ? false : true;
            }
        } else {
            (iscolor == true) ? $(this).addClass("tdbg") : $(this).removeClass("tdbg");
            iscolor = (iscolor == true) ? false : true;
        }
    });
}

/**
 *
 * @param el
 */
function makeExpandingArea(el) {
    var timer = null;
    //由于ie8有溢出堆栈问题，故调整了这里
    var setStyle = function(el, auto) {
        if (auto) el.style.height = 'auto';
        if(el.scrollHeight<70){
            el.style.height = '70px';
        }else{
            el.style.height = el.scrollHeight + 'px';
        }
    }
    var delayedResize = function(el) {
        if (timer) {
            clearTimeout(timer);
            timer = null;
        }
        timer = setTimeout(function() {
            setStyle(el)
        }, 200);
    }
    if (el.addEventListener) {
        el.addEventListener('input', function() {
            setStyle(el, 1);
        }, false);
        setStyle(el)
    } else if (el.attachEvent) {
        el.attachEvent('onpropertychange', function() {
            setStyle(el)
        })
        setStyle(el)
    }
    if (window.VBArray && window.addEventListener) { //IE9
        el.addEventListener("onkeydown", function() {
            var key = window.event.keyCode;
            if (key == 8 || key == 46) delayedResize(el);

        });
        el.addEventListener("oncut", function() {
            delayedResize(el);
        }); //处理粘贴
    }
    $(el).css("overflow-y", "hidden");
    _resizeIframe();
}

//底部按钮区固定,分几种情况进行处理
function bottomFix() {
    var btgroup = $(".container .bt-group:not(.nofix)");
    var dh = top.$(".sl-header").outerHeight() + Math.floor(top.$(".sl-mainpanel>.sl-pagebody>.container").outerHeight());
    var st = $(this).scrollTop(); //滚动距离
    var wh = $(window).height(); //窗口高度
    var vh = st + wh; //已视总高度
    if(vh < dh){
        btgroup.addClass("fixed");
    } else {
        btgroup.removeClass("fixed");
    }
}
function bottomFixLayer() {
    var btgroup = $(".container-popup .bt-group:not(.nofix)");
    var dh = $(".container-popup").outerHeight();
    var st = $(this).scrollTop(); //滚动距离
    var wh = $(window).height(); //窗口高度
    var vh = st + wh; //已视总高度
    if(vh < dh){
        btgroup.addClass("fixed");
    } else {
        btgroup.removeClass("fixed");
    }
}
function bottomFixIframe() {
    var btgroup = $(".container-embed .bt-group:not(.nofix)");
    var st = $(top).scrollTop(); //滚动距离
    var wph = $(window.parent).height(); //窗口高度
    var wot = $(window.parent.document).find(frameElement).offset().top; //iframe上边距
    var wsh = $(window).height(); //iframe自身高度
    var total = wot + wsh; //内容实际总高度
    if(total-wph-st > 0){
        btgroup.addClass("fixed");
        var newposition = total-wph-st > 0 ? total-wph-st : 0 ;
        btgroup.css("bottom",newposition+"px");
    } else {
        btgroup.removeClass("fixed");
        btgroup.css("bottom","0");
    }
}

$(function () {

        if (self == top && self.frames.length == 0 && ($(".container .bt-group:not(.nofix)").length>0)){
            //常规普通页
