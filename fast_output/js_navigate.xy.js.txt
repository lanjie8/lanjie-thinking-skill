/**
 * 提交当前页面
 * @param options
 */
function submitThis(options) {
    if (typeof options == "undefined" || typeof options.formid == "undefined" || !options.formid) {
        formx = $("#isFmVisibleForm").parent("form");
    } else {
        formx = $("#" + options.formid);
    }
    opContent(formx);
    var defaults = {
        formid:'',
        method:'post'
    };
    options = $.extend(defaults,options);

    var defaultMethod = formx.attr("method");
    formx.attr("method", options.method);
    //获取url请求的参数
    var urlParams = getUrlParam();

    $(urlParams).each(function(){
        var urlParam = this.toString();
        var formParam = formx.find("[name="+urlParam+"]");
        var useUrl = true;
        if(formx.find("[id="+urlParam+"UseUrl]").length>0 && formx.find("[id="+urlParam+"UseUrl]")[0].val()!=="true"){
            useUrl = false
        }
        if(formParam.length>0 && useUrl ){
            formParam.remove();
        }
    });

    $("#serializeQuery").val("");
    var input = $("<input type='hidden'>");
    input.attr({"name": "trt"});
    input.val($('meta[name=xy_trt]').attr('content'));
    formx.append(input);
    formx.submit();
    //恢复现场
    formx.attr("method", defaultMethod);
}
function getUrlParam() {
    var url = location.search; //获取url中"?"符后的字串
    var theRequest = [];
    if (url.indexOf("?") != -1) {
        var str = url.substr(1);
        var strs = str.split("&");
        for(var i = 0; i < strs.length; i ++) {
            theRequest[i]=strs[i].split("=")[0];
        }
    }
    return theRequest;
}

/**
 * 用于替换页面后返回上一页
 * @param url
 * @param queryCriteria 用于第二个列表页跳转回去的时候，附带第二个列表页的查询条件的
 * @param backOrClose 如果为true时弹出框的back执行shutCors
 */
function back(url,queryCriteria,backOrClose,parametersBs) {
    if(typeof backOrClose == "undefined" ){
        backOrClose = false;
    }
    try{
        if(backOrClose && this.window.name.indexOf("layui-layer-iframe") == 0) {
            shutCors();
        }
    } catch (e) {
        console.log(e);
    }
    if(typeof queryCriteria == "undefined" ){
        queryCriteria = true
    }
    var formx = $("<form method='post'></form>");
    var input;
    if(typeof url == "undefined" || url==null || url==""){
        url=$("#previousUrl").val();
    }
    if(url.indexOf("trt") == -1) {
        if (url.indexOf("?") > 0) {
            url = url + "&trt=" + $('meta[name=xy_trt]').attr('content');
        } else {
            url = url + "?trt=" + $('meta[name=xy_trt]').attr('content');
        }
    }
    formx.attr({"action": url});
    // input = $("<input type='hidden'>");
    // input.attr({"name": "trt"});
    // input.val($('meta[name=xy_trt]').attr('content'));
    // formx.append(input);
    if(parametersBs){
        var serializeQuery = decodeURIComponent($.base64.atob(localStorage.getItem(parametersBs),false));
        var serializeQueryJson = parseQueryString (serializeQuery);
        var urlQueryString = parseQueryString(url);
        $.each(serializeQueryJson, function (key, value) {
            if(!urlQueryString.hasOwnProperty(key)){
                input = $("<input type='hidden'>");
                input.attr({"name": key});
                input.val(value);
                formx.append(input);
            }
        });
        input = $("<input type='hidden'>");
        input.attr({"name": "_csrf"});
        input.val($('meta[name=_csrf]').attr('content'));
        formx.append(input);
    }else{
        if (typeof $("#serializeQuery").val() != "undefined" && $("#serializeQuery").val() != "" && queryCriteria) {
            if(queryCriteria){
                var serializeQuery = decodeURIComponent($.base64.atob($("#serializeQuery").val(), true));
                var serializeQueryJson = parseQueryString (serializeQuery);
                var urlQueryString = parseQueryString(url);
                $.each(serializeQueryJson, function (key, value) {
                    if(!urlQueryString.hasOwnProperty(key)){
                        input = $("<input type='hidden'>");
                        input.attr({"name": key});
                        input.val(value);
                        formx.append(input);
                    }
                });
            }
        } else {
            input = $("<input type='hidden'>");
            input.attr({"name": "_csrf"});
            input.val($('meta[name=_csrf]').attr('content'));
            formx.append(input);
        }
    }

    formx.appendTo(document.body);
    formx.submit();
}

/**
 * 替换页面跳转方法
 * @param obj 可以是button或者a的this，也可以使{action:''}
 * @async 是否为异步操作
 */
function ajaxDefault(obj,obj2,obj3,obj4) {
    var async = false;
    if("boolean" ==typeof obj3){
        async = obj3;
    }
    if("boolean" == typeof obj4){
        async = obj4;
    }
    if(isElement(obj)){
        return ajaxDefaultDom(obj,obj2,async);
    }else{
        return ajaxDefaultPara(obj,obj2,obj3,async);
    }
}

function ajaxDefaultDom(obj,doneFn,async) {
    var url = $(obj).attr("url");
    var parameters = $(obj).attr("parameters");
    var trt =$(obj).attr("trt");
    if(!parameters){
       parameters = {}
    }
    if(trt){
        if (jQuery.type(parameters) == "string") {
            parameters = eval("(" + parameters + ")");
        }
        parameters.trt = $(obj).attr("trt");
    }
    return ajaxDefaultPara(url,parameters,doneFn,async);
}

/**
 * 通用的ajax提交方法
 * @param url 请求地址
 * @param parameters 形如{"key":"value"}的数据
 * @param async 是否异步
 */
//必须写在这才能返回
var returnValue;

function ajaxDefaultPara(url, parameters, doneFn,async) {
    var parametersString = "";
    if (parameters) {
        if (jQuery.type(parameters) == "string") {
            parameters = eval("(" + parameters + ")");
        }

        $.each(parameters, function (key, value) {
            //encodeURIComponent处理个别情况乱码问题
            parametersString = parametersString + "&" + key + "=" + encodeURIComponent(value);
        });
    }
    var csrftoken = $('meta[name=_csrf]').attr('content');
    $.ajax({
        method: "POST",
        url: url,
        async: async,
        headers: {
            accept: 'application/json',
            'X-CSRF-TOKEN': csrftoken
        },
        data: parametersString,
        beforeSend: function () {
            processIndex = layer.load();
        }
    })
        .done(function (returnValueAjax) {
            layer.close(processIndex);
            if (doneFn) {
                doneFn(returnValueAjax);
            }
            returnValue = returnValueAjax;
        }).error(function (error) {
        layer.close(processIndex);
        console.log(JSON.stringify(error));
    });
    return returnValue;
}

/**
 * 页面跳转方法，在a或者button按钮上使用
 * @param button button或者a元素本身
 * @param options 选项
 */
function forwardDom(button,options) {
    if (typeof options == "undefined" ) {
        options={};
    }
    if(!options.action  && options.url){
        options.action = options.url;
    }
    if(!options.action  && $(button).attr("url")){
        options.action = $(button).attr("url");
    }
    if($(button).attr("method")){
        options.method = $(button).attr("method");
    }
    if($(button).attr("target")){
        options.target = $(button).attr("target");
    }
    if($(button).attr("formid")){
        options.formid = $(button).attr("formid");
    }
    if($(button).attr("trt")){
        options.trt = $(button).attr("trt");
    }
    if($(button).attr("parameters")){
        options.parameters = $(button).attr("parameters");
    }
    if($(button).attr("parametersBs")){
        options.parametersBs = $(button).attr("parametersBs");
    }

    forwardParas(options);
}

/**
 * 页面跳转方法，用在js中forwardParas({action:''})
 * @param options
 */
function forwardParas(options) {
    var defaults = {
        queryCriteria: true,
        method: 'get',
        target:'_self',
        exportExcel:false,
        exportExcelAppendQuery:false,
    };
    options = $.extend(defaults,options);
    //此处调用确保属性合并完成
    //如果是自己替换不加消失时间
    if("_self" === options.target && !options.exportExcel){
        layer.load();
    }else{
        layer.load(null, {time: 3000});
    }
    var formOragin;
    if (options.formid) {
        formOragin = $("#" + options.formid);
    } else {
        formOragin = $("#isFmVisibleForm").parent("form");
    }

    var formSubmit = $("<form></form>");

    //清理xytable.js中生成的动态列数据
    if(!options.exportExcel){
        $("#exportColumns").val("");
    }else{
        var inputP = $("<input type='hidden'>");
        inputP.attr({"name": "exportColumns"});
        inputP.attr({"id": "exportColumns"});
        inputP.val($("#exportColumns").val());
        formSubmit.append(inputP);
        //导出excel时自动追加查询条件
        if(options.exportExcelAppendQuery){
            var urlQueryString = parseQueryString(decodeURI(formOragin.serialize()));
            $.each(urlQueryString, function (key, value) {
                if(key!="exportColumns"){
                    input = $("<input type='hidden'>");
                    input.attr({"name": key});
                    input.val(value);
                    formSubmit.append(input);
                }
            });
        }
    }

    $("#serializeQuery").val("");
    if (options.queryCriteria) {
        var serializeQuery = $.base64.btoa(formOragin.serialize(), true);
        var inputQ = $("<input type='hidden'>");
        inputQ.attr({"name": "serializeQuery"});
        inputQ.attr({"id": "serializeQuery"});
        inputQ.val(serializeQuery);
        formSubmit.append(inputQ);
    }
    if(options.parametersBs){
        localStorage.setItem(options.parametersBs, $.base64.btoa(formOragin.serialize(), true));
    }
    if (options.trt) {
        var inputT = $("<input type='hidden'>");
        inputT.attr({"name": "trt"});
        inputT.attr({"id": "trt"});
        inputT.val(options.trt);
        formSubmit.append(inputT);
    }

    addBizInputs(formOragin, formSubmit);

    if(options.method=="post"){
        var inputCsrf = $("<input type='hidden'>");
        inputCsrf.attr({"name": "_csrf"});
        inputCsrf.attr({"id": "_csrf"});
        inputCsrf.val($('meta[name=_csrf]').attr('content'));
        formSubmit.append(inputCsrf);
    }

    genParameters(options, formSubmit);

    formSubmit.attr("method", options.method);
    formSubmit.attr("target", options.target);
    formSubmit.appendTo(document.body);
    formSubmit.attr("action", options.action).submit();
}

/**
 * 私有方法
 * @param options
 */
function openWindowParas(options) {
    var defaults = {
        title: '',
        size: 'nomal',
        max: false,
        closeBtn: 1,
        showTitle:true,
        maxmin:true
    };
    options = $.extend(defaults,options);
    var formSubmit = $("<form></form>");
    //数据权限
    if (options.trt) {
        var inputT = $("<input type='hidden'>");
        inputT.attr({"name": "trt"});
        inputT.attr({"id": "trt"});
        inputT.val(options.trt);
        formSubmit.append(inputT);
    }
    genParameters(options, formSubmit);
    //当前窗口所在frameName，弹出窗口回调使用
    var input = $("<input type='hidden'>");
    input.attr({"name": "frameName"});
    input.attr({"id": "frameName"});
    if (options.framepath && options.framepath != "") {
        input.val(options.framepath);
    } else if (window.name) {
        input.val("top['" + window.name + "']");
    }
    formSubmit.append(input);
    //from=http...跨域访问使用
    formSubmit.append(genFrom());

    if(options.action.indexOf("?") > 0) {
        options.content = options.action + "&" + formSubmit.serialize();
    } else {
        options.content = options.action + "?" + formSubmit.serialize();
    }

    sendMessageCrossDomain("popup", options);
}

/**
 * 获取options中的parameters参数，并形成input追加到form
 * @param options
 * @param formSubmit
 */
function genParameters(options, formSubmit) {
//参数
    if (options.parameters) {
        if (typeof options.parameters == "string") {
            options.parameters = eval("(" + options.parameters + ")");
        }

        $.each(options.parameters, function (key, value) {
            var input = $("<input type='hidden'>");
            input.attr({"name": key});
            input.attr({"id": key});
            input.val(value);
            formSubmit.append(input);
        });
    }
}

//带参数提交不带跳转
function holdParas(url, doneFn, formid) {
    var parametersString = "";

    if (formid) {
        parametersString = $('#' + formid).serialize();
    } else {
        parametersString = $("#isFmVisibleForm").parent("form").serialize();
    }

    $.ajax({
        method: "POST",
        url: url,
        headers: {accept: 'application/json'},
        data: parametersString,
        beforeSend: function () {
            // processIndex = layer.load(null,{shade: [0.8, '#393D49']});
            var datas = {
                'iframeName':this.name,
            };
            parentXmxdPopup('loadOpenType',datas);
        }
    })
        .done(function (returnValue) {
            var datas = {
                'iframeName':this.name,
            };
            parentXmxdPopup('loadCloseType',datas);
            doneFn(returnValue);
        })
        .error(function(jqXHR, textStatus, errorThrown){
            alert(jqXHR.status + ":" + jqXHR.responseText);
            parentXmxdPopup('loadCloseType',datas);
        });
}

    });
}
function opContent(form){
